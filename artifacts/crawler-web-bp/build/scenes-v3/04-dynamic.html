import requests, json
students = []
for page in range(1, 5):            # 1..4，第 5 页取不到
    url = "http://127.0.0.1:5000/data?page=" + str(page)   # :5000 端口示意
    r = requests.get(url)
    r.encoding = "utf-8"
    data = json.loads(r.text)       # 字符串 → 字典（s = string）
    for item in data["data"]:       # 顶层 key 叫 data，值是列表
        students.append(item.strip())
讲师原话：「你观察一下这个结构，它是有一个 key 叫 data，这个 data 的 value 呢是一个列表」。URL 里的 :5000 为端口示意（讲师现场改过端口）。
